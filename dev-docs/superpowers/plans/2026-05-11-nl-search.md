```swift
import Foundation
#if canImport(FoundationModels)
import FoundationModels
#endif

#if canImport(FoundationModels)
@Generable
public struct AvailabilityIntent: Codable, Sendable {
    @Guide(description: "Start date (YYYY-MM-DD) of the range the user is asking about. Resolve relative phrases like 'next Tuesday' or 'end of next month' against the current date provided in instructions.")
    public var startDate: String?

    @Guide(description: "End date (YYYY-MM-DD) of the range, inclusive. Equal to startDate for a single day. Nil if no date was mentioned.")
    public var endDate: String?

    @Guide(description: "Earliest hour the slot may start (0-23). E.g., 'after 2pm' -> 14, 'morning' -> 8.")
    public var startHour: Int?

    @Guide(description: "Latest hour the slot may end (1-24). E.g., 'before noon' -> 12, 'afternoon' -> 17.")
    public var endHour: Int?

    @Guide(description: "Minimum free block length in minutes, if user mentioned a duration ('30 min', 'an hour', 'quick 15').")
    public var durationMin: Int?

    @Guide(description: "Short free-text purpose if user gave one ('coffee', 'a call with Anna'). Optional, for display only.")
    public var purposeHint: String?

    public init() {}
}
#else
public struct AvailabilityIntent: Codable, Sendable {
    public var startDate: String?
    public var endDate: String?
    public var startHour: Int?
    public var endHour: Int?
    public var durationMin: Int?
    public var purposeHint: String?
    public init() {}
}
#endif

public extension AvailabilityIntent {

    /// The concrete date span the user asked about, or nil if the
    /// query didn't mention a date at all. Single-day queries return
    /// (start, end) where both fall on the same calendar day.
    func requestedRange(now: Date) -> (start: Date, end: Date)? {
        guard let startStr = startDate, let start = Self.parse(date: startStr) else {
            return nil
        }
        let end: Date
        if let endStr = endDate, let parsed = Self.parse(date: endStr) {
            end = parsed
        } else {
            end = start
        }
        return (start, end)
    }

    /// Human-readable summary used by `QueryResultBanner`.
    /// Example: "TUE APR 21 · AFTER 2PM · 60 MIN · COFFEE".
    func summary(now: Date) -> String {
        var parts: [String] = []

        if let range = requestedRange(now: now) {
            parts.append(Self.dateLabel(start: range.start, end: range.end, now: now))
        }

        if let lo = startHour, let hi = endHour {
            parts.append("\(Self.hourLabel(lo))–\(Self.hourLabel(hi))")
        } else if let lo = startHour {
            parts.append("AFTER \(Self.hourLabel(lo))")
        } else if let hi = endHour {
            parts.append("BEFORE \(Self.hourLabel(hi))")
        }

        if let mins = durationMin {
            parts.append("\(mins) MIN")
        }

        if let purpose = purposeHint, !purpose.isEmpty {
            parts.append(purpose.uppercased())
        }

        return parts.joined(separator: " · ")
    }

    // MARK: - Internal helpers (visible for testing)

    static func parse(date: String) -> Date? {
        let f = DateFormatter()
        f.dateFormat = "yyyy-MM-dd"
        f.timeZone = .current
        return f.date(from: date)
    }

    static func dateLabel(start: Date, end: Date, now: Date) -> String {
        let cal = Calendar.current
        let isRange = !cal.isDate(start, inSameDayAs: end)

        if !isRange {
            if cal.isDate(start, inSameDayAs: now) { return "TODAY" }
            if let tomorrow = cal.date(byAdding: .day, value: 1, to: now),
               cal.isDate(start, inSameDayAs: tomorrow) { return "TOMORROW" }
            return Self.shortDate(start)
        }
        return "\(Self.shortDate(start))–\(Self.shortDate(end))"
    }

    static func shortDate(_ d: Date) -> String {
        let f = DateFormatter()
        f.dateFormat = "EEE MMM d"
        return f.string(from: d).uppercased()
    }

    /// 14 -> "2PM", 12 -> "12PM", 0 -> "12AM", 9 -> "9AM".
    static func hourLabel(_ hour: Int) -> String {
        let h = hour % 24
        if h == 0  { return "12AM" }
        if h == 12 { return "12PM" }
        if h < 12  { return "\(h)AM" }
        return "\(h - 12)PM"
    }
}
```

`OpenTimeUI/Sources/OpenTimeUI/Query/QueryFilter.swift`:

```swift
import Foundation

public enum QueryFilter {

    /// Strict filter. Returns only days/slots that satisfy every populated
    /// field of `intent`. Past slots (start <= now) are always excluded.
    public static func filter(
        days: [DayAvailability],
        intent: AvailabilityIntent,
        now: Date
    ) -> [DayAvailability] {

        let cal = Calendar.current
        let dateRange = intent.requestedRange(now: now)

        var out: [DayAvailability] = []

        for day in days {
            // Date filter — skip days outside the requested range.
            if let range = dateRange {
                let startDay = cal.startOfDay(for: range.start)
                let endDay   = cal.startOfDay(for: range.end)
                let thisDay  = cal.startOfDay(for: day.date)
                if thisDay < startDay || thisDay > endDay { continue }
            }

            // Per-slot pipeline: drop past, intersect with time bounds, check duration.
            var keptSlots: [FreeSlot] = []
            for raw in day.freeSlots {
                guard let trimmed = intersect(slot: raw, intent: intent, dayStart: cal.startOfDay(for: day.date)) else { continue }
                if trimmed.end <= now { continue }
                let startOrNow = max(trimmed.start, now)
                let usable = FreeSlot(start: startOrNow, end: trimmed.end)
                if let minDur = intent.durationMin, usable.durationMin < minDur { continue }
                keptSlots.append(usable)
            }

            guard !keptSlots.isEmpty else { continue }
            out.append(DayAvailability(date: day.date, freeSlots: keptSlots, busySlots: day.busySlots))
        }

        return out
    }

    /// Intersect `slot` with the `startHour..endHour` window on the same calendar day.
    /// Returns nil if either bound makes the window empty.
    private static func intersect(slot: FreeSlot, intent: AvailabilityIntent, dayStart: Date) -> FreeSlot? {
        let cal = Calendar.current
        var lo = slot.start
        var hi = slot.end

        if let h = intent.startHour {
            let bound = cal.date(byAdding: .hour, value: h, to: dayStart)!
            if bound > lo { lo = bound }
        }
        if let h = intent.endHour {
            let bound = cal.date(byAdding: .hour, value: h, to: dayStart)!
            if bound < hi { hi = bound }
        }

        return lo < hi ? FreeSlot(start: lo, end: hi) : nil
    }
}
```

```swift
import Foundation

public enum QuerySuggest {

    public struct Suggestion: Sendable {
        public let slot: FreeSlot
        public let day: Date
        public let relaxedFields: [RelaxedField]
    }

    public enum RelaxedField: Sendable {
        case date, timeOfDay, duration
    }

    /// Called only when `QueryFilter.filter` returns empty. Tries relaxing
    /// one constraint at a time in this priority order: timeOfDay → duration → date.
    /// Returns the earliest matching slot, or nil if nothing even loosely fits.
    /// Never returns a slot before the requested date.
    public static func closest(
        days: [DayAvailability],
        intent: AvailabilityIntent,
        now: Date
    ) -> Suggestion? {

        // Step 1: relax timeOfDay (drop startHour/endHour).
        var s1 = intent
        s1.startHour = nil
        s1.endHour   = nil
        if let hit = earliest(days: days, intent: s1, now: now, requestedDate: intent.requestedRange(now: now)?.start) {
            return Suggestion(slot: hit.slot, day: hit.day, relaxedFields: [.timeOfDay])
        }

        // Step 2: relax duration (keep date + time bounds).
        var s2 = intent
        s2.durationMin = nil
        if let hit = earliest(days: days, intent: s2, now: now, requestedDate: intent.requestedRange(now: now)?.start) {
            return Suggestion(slot: hit.slot, day: hit.day, relaxedFields: [.duration])
        }

        // Step 3: relax date (keep time + duration). Search forward from
        // the requested date — never before it.
        var s3 = intent
        s3.startDate = nil
        s3.endDate   = nil
        if let hit = earliest(days: days, intent: s3, now: now, requestedDate: intent.requestedRange(now: now)?.start) {
            return Suggestion(slot: hit.slot, day: hit.day, relaxedFields: [.date])
        }

        return nil
    }

    private struct Hit { let slot: FreeSlot; let day: Date }

    private static func earliest(
        days: [DayAvailability],
        intent: AvailabilityIntent,
        now: Date,
        requestedDate: Date?
    ) -> Hit? {
        let filtered = QueryFilter.filter(days: days, intent: intent, now: now)
        let cal = Calendar.current
        let lowerBound = requestedDate.map { cal.startOfDay(for: $0) }

        for day in filtered.sorted(by: { $0.date < $1.date }) {
            if let lb = lowerBound, cal.startOfDay(for: day.date) < lb { continue }
            if let slot = day.freeSlots.sorted(by: { $0.start < $1.start }).first {
                return Hit(slot: slot, day: day.date)
            }
        }
        return nil
    }
}
```

```swift
        #if canImport(FoundationModels)

        let response = try await session.respond(to: text, generating: AvailabilityIntent.self)

        #else

        #endif

    static func instructions(now: Date) -> String {

        let weekdayFmt = DateFormatter()
        weekdayFmt.dateFormat = "EEEE"
        let weekday = weekdayFmt.string(from: now)

        You parse a user's free-form question about their calendar availability \
        into a structured intent. Today is \(today) (\(weekday)).

        Rules:
        - startDate / endDate: absolute YYYY-MM-DD dates. Resolve "next Tuesday", \
        "this weekend", "end of next month", etc. against today's date. \
        Ambiguous weekdays mean the NEXT future occurrence. Single-day queries set both fields equal.
        - "This week" = today through the coming Sunday.
        - "Next week" = Monday through Sunday of the following week.
        - "Weekend" = Saturday–Sunday.
        - startHour / endHour: 24-hour integers. "Morning" = 8–12, "afternoon" = 12–17, \
        "evening" = 17–22. "After 2pm" -> startHour 14. "Before noon" -> endHour 12.

        - If the user said no date at all, leave startDate and endDate nil.
```

```swift
func extendRange(to target: Date) async {
```

```swift
@MainActor
private func runQuery() async {
    let text = queryText.trimmingCharacters(in: .whitespaces)
    guard !text.isEmpty else { return }
    queryIsLoading = true
    queryError = false
    defer { queryIsLoading = false }

    do {
        let now = Date()
        let intent = try await QueryParser.parse(text, now: now)

        // Auto-expand the loaded date range up to the 4-week cap.
        if let needed = intent.requestedRange(now: now) {
            let cap = Calendar.current.date(byAdding: .weekOfYear, value: 4, to: now)!
            let target = min(needed.end, cap)
            if let currentEnd = manager.loadedRangeEnd, target > currentEnd {
                await manager.extendRange(to: target)
            }
        }

        let matched = QueryFilter.filter(days: filteredDays, intent: intent, now: now)
        queryIntent = intent
        queryResult = matched
        querySuggestion = matched.isEmpty
            ? QuerySuggest.closest(days: filteredDays, intent: intent, now: now)
            : nil
    } catch {
        queryError = true
        queryResult = nil
        queryIntent = nil
        querySuggestion = nil
    }
}

private func clearQuery() {
    queryText = ""
    queryResult = nil
    queryIntent = nil
    querySuggestion = nil
    queryError = false
}

private func bannerState(for intent: AvailabilityIntent) -> QueryResultBanner.State {
    let summary = intent.summary(now: Date())
    if let matches = queryResult, !matches.isEmpty {
        let count = matches.reduce(0) { $0 + $1.freeSlots.count }
        return .match(summary: summary, slotCount: count)
    }
    if let s = querySuggestion {
        return .suggestion(
            summary: summary,
            suggestionText: Self.formatSuggestion(s),
            onTap: { deepLink.pendingSlot = s.slot }
        )
    }
    return .noMatch(summary: summary)
}

private static func formatSuggestion(_ s: QuerySuggest.Suggestion) -> String {
    let f = DateFormatter()
    f.dateFormat = "EEE h:mm a"
    let when = f.string(from: s.slot.start).uppercased()
    let relaxed = s.relaxedFields.map { Self.relaxLabel($0) }.joined(separator: ", ")
    return "\(when) (\(relaxed) RELAXED)"
}

private static func relaxLabel(_ field: QuerySuggest.RelaxedField) -> String {
    switch field {
    case .date:      return "DATE"
    case .timeOfDay: return "TIME"
    case .duration:  return "DURATION"
    }
}
```
